# pip constraints for the macOS .app build (pip install -c this-file ...).
#
# Every prebuilt wheel carries its own minimum macOS; MACOSX_DEPLOYMENT_TARGET
# does not lower it. Left unpinned, pip on a current build host picks
# onnxruntime 1.23.2 (Intel, needs macOS 13) or >= 1.24 (Apple Silicon, needs
# macOS 14), which silently raises the whole app's minimum macOS.
# 1.19.2 is the newest cp312 release tagged macosx_11_0 (universal2) and was
# verified end-to-end (VAD + transcription) on macOS 10.15 in 2026-09.
# See docs/MACOS_BUILD_NOTES.md.
onnxruntime==1.19.2
# PyAV 19.0.0 (2026-09-29) removed av.open(metadata_errors=...), which
# faster-whisper <= 1.2.1 still passes: with av 19 the frozen app fails every
# transcription ("open() got an unexpected keyword argument
# 'metadata_errors'"; CI smoke test, 2026-09-30). Same pin as requirements.txt.
av<19
# numpy comes in through faster-whisper / ctranslate2 / onnxruntime. Unpinned,
# each build took the newest release, whose wheels can carry a higher minimum
# macOS than the rest of the bundle. 1.26.4 is the version the macOS test
# workflows (macos-e2e, macos-compileall-script-test) install.
numpy==1.26.4
# The build tool itself, so two builds of the same commit bundle the same way.
pyinstaller==6.22.3
